# Runtime dependencies for the Streamlit app.
# Streamlit Community Cloud installs from this file, so it is kept to what the
# app actually imports. Training and notebook extras live in requirements-dev.txt.
#
# Deployment notes, each of which was an actual deploy failure:
#
#  * Set the Python version to 3.12 in the app's Advanced settings. Streamlit
#    Cloud defaulted to 3.14, for which TensorFlow publishes no wheels at all,
#    and the build failed with "no wheels with a matching Python ABI tag".
#  * tensorflow-cpu rather than tensorflow: the default wheel bundles CUDA
#    libraries worth gigabytes that a CPU-only host can never use.
#  * torch from the PyTorch CPU index for the same reason; the PyPI default is
#    the CUDA build.
#  * opencv-python-headless rather than opencv-python: the server has no libGL,
#    and the GUI build fails at import with "libGL.so.1: cannot open shared
#    object file".

--extra-index-url https://download.pytorch.org/whl/cpu

# Detection and classification
tensorflow-cpu>=2.18.0
# Keras 3.11 cannot deserialize models saved by a newer Keras: it rebuilds
# EfficientNet's stem with one Rescaling layer where the saved graph has two.
# Kaggle trained these on 3.13.2.
keras>=3.13.0

# Segmentation
torch>=2.2.0
torchvision>=0.17.0
segmentation-models-pytorch>=0.3.3

# Preprocessing and display
opencv-python-headless>=4.9.0
Pillow>=10.2.0
numpy>=1.26.0
matplotlib>=3.8.0
PyYAML>=6.0.1

streamlit>=1.33.0
